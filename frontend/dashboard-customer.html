${statusBadge(o.status)} ${payStatusBadge(o.payment_status || 'unpaid')}

Belum ada pesanan.

'; } function renderPaymentSection(o, qrisUrl) { if (o.payment_status === 'paid') { return `

`; } if (o.payment_status === 'verifying') { return `

`; } return `

💳 Pembayaran QRIS

${qrisUrl ? `` : '

QRIS belum diatur oleh Admin.
